\section{Installation}

Install RateKing using pip:

\begin{lstlisting}[language=bash, frame=single]
pip install RateKing
\end{lstlisting}

Verify installation:

\begin{lstlisting}
import RateKing
print(RateKing.__version__)  # Should print: 0.1.0
\end{lstlisting}

\section{Basic Example: NH$_3$ + OH $\to$ NH$_2$ + H$_2$O}

\subsection{Step 1: Import Required Modules}

\begin{lstlisting}
from RateKing import Reaction, TST
from RateKing.tunneling_corrections import eckartCorrection
\end{lstlisting}

\subsection{Step 2: Define Reaction Components}

\begin{lstlisting}
# Reactants
reactant1 = {
    "name": "NH3",
    "filePath": "path/to/nh3.out",
    "link": 2  # Or -1 for last link
}

reactant2 = {
    "name": "OH",
    "filePath": "path/to/oh.out",
    "link": 2
}

# Transition State
ts = {
    "name": "TS",
    "filePath": "path/to/ts.out",
    "link": 2
}

# Products
product1 = {
    "name": "NH2",
    "filePath": "path/to/nh2.out",
    "link": 2
}

product2 = {
    "name": "H2O",
    "filePath": "path/to/h2o.out",
    "link": 2
}
\end{lstlisting}

\subsection{Step 3: Create and Run Reaction}

\begin{lstlisting}
# Create reaction object
reaction = Reaction(
    name="NH3+OH",
    reactantsPath=[reactant1, reactant2],
    productsPath=[product1, product2],
    transitionStatePath=ts,
    temperature=298.15
)

# Calculate reaction properties
reaction.run()
\end{lstlisting}

\subsection{Step 4: Calculate Rate Constants}

\begin{lstlisting}
# TST rate constant
tst = TST(reaction)
k_tst = tst.calculateKTST(bi_or_uni='bi')

# Apply Eckart tunneling correction
eckart = eckartCorrection(reaction)
k_eckart = k_tst * eckart.kappaEckart

print(f"k_TST    = {k_tst:.3e} cm^3/molecule/s")
print(f"k_Eckart = {k_eckart:.3e} cm^3/molecule/s")
print(f"Tunneling correction factor = {eckart.kappaEckart:.2f}")
\end{lstlisting}

\section{Temperature Range Calculation}

\begin{lstlisting}
import numpy as np

# Define temperature range
temperatures = np.arange(200, 1001, 100)

# Calculate for each temperature
for T in temperatures:
    rxn = Reaction(..., temperature=T)
    rxn.run()
    
    tst = TST(rxn)
    k = tst.calculateKTST()
    
    print(f"T = {T} K: k = {k:.3e}")
\end{lstlisting}

\section{Available Tunneling Corrections}

\begin{lstlisting}
from RateKing.tunneling_corrections import (
    eckartCorrection,    # Most accurate
    wignerCorrection,    # Simplest, low T
    bell35Correction,    # Bell (1935)
    bell58Correction,    # Bell (1958)
    bell2TCorrection     # Bell 2T
)

# Apply any correction
correction = eckartCorrection(reaction)
kappa = correction.kappaEckart
k_corrected = k_tst * kappa
\end{lstlisting}

\section{Common Tasks}

\subsection{Check Available Links in File}

\begin{lstlisting}
from RateKing import Molecule

total_links = Molecule.getAvailableLinks("file.out")
print(f"File contains {total_links} links")
\end{lstlisting}

\subsection{Access Reaction Properties}

\begin{lstlisting}
reaction.run()

print(f"Ea = {reaction.activationEnergy:.2f} kcal/mol")
print(f"Delta E = {reaction.deltaE:.2f} kcal/mol")
print(f"Delta H = {reaction.deltaH:.2f} kcal/mol")
print(f"Delta G = {reaction.deltaG:.2f} kcal/mol")
print(f"Imaginary freq = {reaction.transitionStateImaginaryFrequency:.1f} cm^-1")
\end{lstlisting}

\subsection{Calculate Deformed TST}

\begin{lstlisting}
from RateKing import dTST

dtst = dTST(reaction)
k_dtst = dtst.calculateKDTST(bi_or_uni='bi')
\end{lstlisting}

\section{File Requirements}

Your Gaussian output files must include:
\begin{itemize}
    \item Optimized geometry (Opt calculation)
    \item Frequency analysis (Freq keyword)
    \item Thermochemistry at desired temperature
    \item For TS: exactly one imaginary frequency
\end{itemize}

Example Gaussian input:
\begin{lstlisting}[language=bash, frame=single]
%NProcShared=8
%Mem=16GB
#P M062X/6-311++G(d,p) Opt Freq Temperature=298.15

Title

0 1
[geometry]

\end{lstlisting}

\section{Troubleshooting}

\subsection{Import Error}
\begin{lstlisting}[language=bash, frame=single]
pip install --upgrade RateKing
\end{lstlisting}

\subsection{File Not Found}
Use absolute paths or verify file locations:
\begin{lstlisting}
import os
print(os.path.abspath("your_file.out"))
\end{lstlisting}

\subsection{No Imaginary Frequency}
Verify your TS file:
\begin{itemize}
    \item Is it actually a transition state optimization?
    \item Did the frequency calculation complete?
    \item Is the link number correct?
\end{itemize}
